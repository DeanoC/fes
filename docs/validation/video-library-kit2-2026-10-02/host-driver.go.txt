package main

import (
 "context"
 "flag"
 "fmt"
 "net/http"
 "os"
 "os/signal"
 "path/filepath"
 "syscall"
 "time"
 "github.com/DeanoC/FogCast/fogcast"
 "github.com/DeanoC/FogCast/internal/hostapi"
)

func main() {
 root := flag.String("root", "", "private diagnostic state directory")
 flag.Parse()
 if *root=="" { panic("missing state root") }
 paths := fogcast.Paths{Config:filepath.Join(*root,"config.toml"),Index:filepath.Join(*root,"library.sqlite3"),
 CorePackages:filepath.Join(*root,"core-packages"),Staging:filepath.Join(*root,"staging"),MetadataRoot:filepath.Join(*root,"metadata"),
 UserLibrary:filepath.Join(*root,"library-user.sqlite3"),LibrarySettings:filepath.Join(*root,"library-settings.json"),
 MediaIndex:filepath.Join(*root,"library-media.sqlite3"),MediaCache:filepath.Join(*root,"library-media")}
 service, err := fogcast.Open(context.Background(),paths,nil)
 if err != nil { panic(err) }
 defer service.Close()
 server := &http.Server{Addr:"127.0.0.1:18787",Handler:hostapi.New(service),ReadHeaderTimeout:10*time.Second}
 stopped := make(chan os.Signal,1)
 signal.Notify(stopped,syscall.SIGTERM,os.Interrupt)
 go func(){ <-stopped; ctx,cancel:=context.WithTimeout(context.Background(),15*time.Second); defer cancel(); server.Shutdown(ctx) }()
 fmt.Println("Isolated production service and HTTP API on 127.0.0.1:18787")
 if err:=server.ListenAndServe(); err!=nil && err!=http.ErrServerClosed { panic(err) }
}
