type uxPassGalleryServices struct { games []hostclient.Game; hardware hostclient.HardwareSnapshot }
func (s uxPassGalleryServices) QueryGames(_ context.Context, q hostclient.GameListQuery, max int) ([]hostclient.Game,error) {
 out:=[]hostclient.Game{}
 for _,g:=range s.games {
  if q.Platform!="" && q.Platform!=g.System {continue}
  if q.Q!="" && !strings.Contains(strings.ToLower(g.Title),strings.ToLower(q.Q)) {continue}
  if len(q.CoreIDs)>0 {found:=false; for _,id:=range q.CoreIDs {if id==g.CoreID {found=true}}; if !found {continue}}
  if q.Genre!="" && q.Genre!=g.Genre {continue}
  out=append(out,g); if max>0 && len(out)>=max {break}
 }
 return out,nil
}
