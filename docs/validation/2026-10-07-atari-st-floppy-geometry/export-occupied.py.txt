#!/usr/bin/env python3
"""Export occupied physical wires from the retained #165 scaffold."""
import hashlib
import json
from pathlib import Path
import sys
source = Path(sys.argv[1])
data = source.read_bytes()
expected = "c11e2f071d93de1ce351434a40f8a01a512769d118b32312d51aff422d944a5c"
if hashlib.sha256(data).hexdigest() != expected:
    raise SystemExit("This probe requires the exact issue #165 seed-4 scaffold")
top = json.loads(data)["modules"]["top"]
with Path("occupied.tsv").open("w") as output:
    for name, net in top["netnames"].items():
        route = net.get("attributes", {}).get("ROUTING", "").split(";")
        for wire in route[::3]:
            if wire and not wire.startswith("WIRE."):
                output.write(wire + "\t" + name + "\n")
