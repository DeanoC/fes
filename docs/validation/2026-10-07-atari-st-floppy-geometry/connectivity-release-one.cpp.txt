#include <cyclonev.h>
#include <vector>
#include <queue>
#include <cstdio>
#include <memory>
#include <fstream>
#include <sstream>
#include <string>
#include <unordered_map>
using C=mistral::CycloneV;
int main(){
 std::unique_ptr<C> c(C::get_model("5CSEBA6U23I7"));
 unsigned n=c->rnode_index_count();
 std::vector<std::vector<unsigned>> down(n);
 std::vector<unsigned char> allowed(n,0);
 std::vector<std::pair<uint32_t,uint32_t>> bits;
 for(unsigned i=0;i<n;i++){auto ro=c->ri2ro(i); if(!ro)continue;
  if(!c->rnode_mux_cram_bits(ro->rc(),bits))return 2;
  allowed[i]=1; for(auto b:bits)if(b.first<1769||b.first>=2806||b.second<3442||b.second>=5162)allowed[i]=0;
  for(auto s=ro->sources_begin();s!=ro->sources_end();++s)down[*s].push_back(i);
 }
 unsigned src=c->pnode_to_rnode(C::pnode_coords(C::LAB,24,41,C::FFB0,5,-1));
 unsigned dst=c->rc2ri(C::rnode_coords(c->rnode_type_lookup("GOUT"),25,42,58));
 printf("source=%s index=%u destination=%u\n",c->ri2rc(src).to_string().c_str(),src,dst);
 std::unordered_map<unsigned,std::string> occupied;
 std::ifstream infile("occupied.tsv");std::string line;while(std::getline(infile,line)){auto tab=line.find('\t');std::string wire=line.substr(0,tab);char type[32];unsigned x,y,z;if(sscanf(wire.c_str(),"%31[^.].%u.%u.%u",type,&x,&y,&z)==4){auto ri=c->rc2ri(C::rnode_coords(c->rnode_type_lookup(type),x,y,z));if(ri!=0xffffffff)occupied[ri]=line.substr(tab+1);}}

 std::vector<std::string> candidates;
 for(auto d:down[src])if(allowed[d]&&occupied.count(d))candidates.push_back(occupied[d]);
 for(auto exempt:candidates){
 std::vector<unsigned char> seen(n,0);std::queue<unsigned> q;q.push(src);seen[src]=1;unsigned count=0;
 while(!q.empty()){auto v=q.front();q.pop();count++;if(v==dst)break;for(auto d:down[v])if(!seen[d]&&allowed[d]&&(!occupied.count(d)||occupied[d]==exempt)){seen[d]=1;q.push(d);}}
 printf("allow_one_shell_net=%s reached=%u destination_reachable=%d\n",exempt.c_str(),count,seen[dst]);
 }
}
