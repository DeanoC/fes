// coreScene is one reference picture, in the shape the v1 reference harness
// uses. The Lua pack reads the scene name from room.initial_game_id. Hint is
// the proposed footer hint. A dialog scene applies a native Settings dialog to
// the snapshot after the room has drawn; FooterOverride then draws that
// dialog's hint in the dimmed footer.
type coreScene struct {
	Name           string
	Room           string
	Hint           string
	FooterOverride bool
	Overlay        func(*Snapshot)
}

func dialogRows(rows ...string) []SettingsRow {
	out := make([]SettingsRow, 0, len(rows)/2)
	for i := 0; i+1 < len(rows); i += 2 {
		out = append(out, SettingsRow{ID: fmt.Sprint(i / 2), Label: rows[i], Value: rows[i+1]})
	}
	return out
}

func dialog(title string, index int, status, hint string, rows ...string) func(*Snapshot) {
	return func(s *Snapshot) {
		s.Settings = SettingsSnapshot{Open: true, Title: title, Index: index, Rows: dialogRows(rows...), Status: status, Hint: hint}
	}
}

func sceneFooter(dev gfx.Device, snap Snapshot, labels map[string]gpuTexture, sc coreScene) {
	layout, ok := rooms.HeroPanel(snap.Room.Width, snap.Room.Height)
	if !ok {
		return
	}
	f := layout.Footer
	x, y := snap.Room.OffsetX+f.X, snap.Room.OffsetY+f.Y
	dev.FillRect(gfx.Rect{X: float32(x), Y: float32(y), W: float32(f.W), H: float32(f.H)}, heroBackground)
	dev.FillRect(gfx.Rect{X: float32(x), Y: float32(y), W: float32(f.W), H: 2}, heroDivider)
	margin := layout.Card.X
	textY := y + (f.H-gfx.TextHeight(heroFooterPx))/2
	hintW := min(gfx.MeasureText(sc.Hint, heroFooterPx), f.W-2*margin)
	heroLine(dev, labels, map[string]struct{}{}, x+f.W-margin-hintW, textY, hintW, heroFooterPx, false, heroSecondary, sc.Hint)
}

func dimFooter(dev gfx.Device, snap Snapshot) {
	layout, ok := rooms.HeroPanel(snap.Room.Width, snap.Room.Height)
	if !ok {
		return
	}
	f := layout.Footer
	dev.SetBlend(gfx.BlendAlpha)
	dev.FillRect(gfx.Rect{X: float32(snap.Room.OffsetX + f.X), Y: float32(snap.Room.OffsetY + f.Y),
		W: float32(f.W), H: float32(f.H)}, gfx.Color{R: 8, G: 8, B: 12, A: 180})
	dev.SetBlend(gfx.BlendNone)
}

func TestCoreScreens(t *testing.T) {
	out, packDir := os.Getenv("FES_CORE_SCREENS"), os.Getenv("FES_CORE_SCREENS_PACK")
	if out == "" || packDir == "" {
		t.Skip("docs/ux/references/core-screens capture harness")
	}
	if err := os.MkdirAll(out, 0o755); err != nil {
		t.Fatal(err)
	}
	pack := rooms.LoadPackFS(os.DirFS(packDir), packDir)
	if !pack.Valid() {
		t.Fatal(pack.Err)
	}
	index := rooms.NewIndex([]rooms.Pack{pack})
	for _, size := range [][2]int{{1280, 720}, {1024, 600}} {
		for _, sc := range coreScenes() {
			app := NewApp(nil, size[0], size[1], 100)
			ctx, cancel := context.WithCancel(context.Background())
			cancel()
			app.ctx, app.roomsIndex, app.attractDisabled = ctx, index, true
			rw, rh := app.roomContentSizeLocked(pack.ID)
			inst, err := rooms.New(pack, rooms.Options{Width: rw, Height: rh, Index: index,
				Budget: hardwareRenderTestBudget(), InitialGameID: sc.Room, ReducedMotion: true})
			if err != nil {
				t.Fatal(err)
			}
			app.room = inst
			if err := inst.Load(); err != nil {
				t.Fatal(err)
			}
			for i := 0; i < 10; i++ {
				app.Tick(time.Now())
				time.Sleep(2 * time.Millisecond)
			}
			if err := inst.Err(); err != nil {
				t.Fatalf("%s: %v", sc.Name, err)
			}
			snap := app.Snapshot()
			if sc.Overlay != nil {
				sc.Overlay(&snap)
			}
			dev, err := gfx.NewSoftware(size[0], size[1])
			if err != nil {
				t.Fatal(err)
			}
			labels := map[string]gpuTexture{}
			presentFrame(dev, snap, map[string]gpuTexture{}, labels, false)
			if sc.Hint == "" {
				sc.Hint = snap.Settings.Hint
			}
			if sc.Overlay == nil || sc.FooterOverride {
				sceneFooter(dev, snap, labels, sc)
				if sc.FooterOverride {
					dimFooter(dev, snap)
				}
			}
			path := filepath.Join(out, fmt.Sprintf("%s-%dx%d.png", sc.Name, size[0], size[1]))
			file, err := os.Create(path)
			if err != nil {
				t.Fatal(err)
			}
			err = png.Encode(file, dev.Snapshot())
			if closeErr := file.Close(); err == nil {
				err = closeErr
			}
			if err != nil {
				t.Fatal(err)
			}
			dev.Close()
			inst.Close()
			t.Logf("%s room %dx%d at %d,%d", path, snap.Room.Width, snap.Room.Height, snap.Room.OffsetX, snap.Room.OffsetY)
		}
	}
}
