// v1Scene is one reference picture. The Lua pack reads the scene name from
// room.initial_game_id. Hint replaces the native hero footer hint, because
// the first-run screens propose their own; an empty hint on Home keeps the
// production one. Native overlays (Settings, the folder picker) are applied
// to the snapshot after the room has drawn.
type v1Scene struct {
	Name     string
	Room     string
	Hint     string
	Native   bool // keep the production footer hint
	NoFooter bool
	Overlay  func(*Snapshot)
}

func v1SettingsRows(rows ...string) []SettingsRow {
	out := make([]SettingsRow, 0, len(rows)/2)
	for i := 0; i+1 < len(rows); i += 2 {
		out = append(out, SettingsRow{ID: fmt.Sprint(i / 2), Label: rows[i], Value: rows[i+1]})
	}
	return out
}

func v1Settings(index int, hint string, rows ...string) func(*Snapshot) {
	return func(s *Snapshot) {
		s.Settings = SettingsSnapshot{Open: true, Index: index, Rows: v1SettingsRows(rows...), Hint: hint}
	}
}

func v1Footer(dev gfx.Device, snap Snapshot, labels map[string]gpuTexture, sc v1Scene) {
	layout, ok := rooms.HeroPanel(snap.Room.Width, snap.Room.Height)
	if !ok || sc.Native {
		return
	}
	f := layout.Footer
	x, y := snap.Room.OffsetX+f.X, snap.Room.OffsetY+f.Y
	dev.FillRect(gfx.Rect{X: float32(x), Y: float32(y), W: float32(f.W), H: float32(f.H)}, heroBackground)
	if sc.NoFooter {
		return
	}
	dev.FillRect(gfx.Rect{X: float32(x), Y: float32(y), W: float32(f.W), H: 2}, heroDivider)
	margin := layout.Card.X
	textY := y + (f.H-gfx.TextHeight(heroFooterPx))/2
	hintW := min(gfx.MeasureText(sc.Hint, heroFooterPx), f.W-2*margin)
	heroLine(dev, labels, map[string]struct{}{}, x+f.W-margin-hintW, textY, hintW, heroFooterPx, false, heroSecondary, sc.Hint)
}

func TestV1Renders(t *testing.T) {
	out, packDir := os.Getenv("FES_V1_RENDERS"), os.Getenv("FES_V1_PACK")
	if out == "" || packDir == "" {
		t.Skip("docs/ux/references/v1 capture harness")
	}
	if err := os.MkdirAll(out, 0o755); err != nil {
		t.Fatal(err)
	}
	pack := rooms.LoadPackFS(os.DirFS(packDir), packDir)
	if !pack.Valid() {
		t.Fatal(pack.Err)
	}
	index := rooms.NewIndex([]rooms.Pack{pack})
	for _, size := range [][2]int{{1280, 720}, {1024, 600}} {
		for _, sc := range v1Scenes() {
			app := NewApp(nil, size[0], size[1], 100)
			ctx, cancel := context.WithCancel(context.Background())
			cancel()
			app.ctx, app.roomsIndex, app.attractDisabled = ctx, index, true
			rw, rh := app.roomContentSizeLocked(pack.ID)
			inst, err := rooms.New(pack, rooms.Options{Width: rw, Height: rh, Index: index,
				Budget: hardwareRenderTestBudget(), InitialGameID: sc.Room, ReducedMotion: true})
			if err != nil {
				t.Fatal(err)
			}
			app.room = inst
			if err := inst.Load(); err != nil {
				t.Fatal(err)
			}
			for i := 0; i < 10; i++ {
				app.Tick(time.Now())
				time.Sleep(2 * time.Millisecond)
			}
			if err := inst.Err(); err != nil {
				t.Fatalf("%s: %v", sc.Name, err)
			}
			snap := app.Snapshot()
			if sc.Overlay != nil {
				sc.Overlay(&snap)
			}
			dev, err := gfx.NewSoftware(size[0], size[1])
			if err != nil {
				t.Fatal(err)
			}
			labels := map[string]gpuTexture{}
			presentFrame(dev, snap, map[string]gpuTexture{}, labels, false)
			if sc.Overlay == nil {
				v1Footer(dev, snap, labels, sc)
			}
			path := filepath.Join(out, fmt.Sprintf("%s-%dx%d.png", sc.Name, size[0], size[1]))
			file, err := os.Create(path)
			if err != nil {
				t.Fatal(err)
			}
			err = png.Encode(file, dev.Snapshot())
			if closeErr := file.Close(); err == nil {
				err = closeErr
			}
			if err != nil {
				t.Fatal(err)
			}
			dev.Close()
			inst.Close()
			t.Logf("%s room %dx%d at %d,%d", path, snap.Room.Width, snap.Room.Height, snap.Room.OffsetX, snap.Room.OffsetY)
		}
	}
}
